      var CIRCLE = Math.PI * 2;
      var MOBILE = /Android|webOS|iPhone|iPad|iPod|BlackBerry/i.test(navigator.userAgent)

      function Controls() {
        this.codes  = { 37: 'left', 39: 'right', 38: 'forward', 40: 'backward' };
        this.states = { 'left': false, 'right': false, 'forward': false, 'backward': false };
        document.addEventListener('keydown', this.onKey.bind(this, true), false);
        document.addEventListener('keyup', this.onKey.bind(this, false), false);
        document.addEventListener('touchstart', this.onTouch.bind(this), false);
        document.addEventListener('touchmove', this.onTouch.bind(this), false);
        document.addEventListener('touchend', this.onTouchEnd.bind(this), false);
      }

      Controls.prototype.onTouch = function(e) {
        var t = e.touches[0];
        this.onTouchEnd(e);
        if (t.pageY < window.innerHeight * 0.5) this.onKey(true, { keyCode: 38 });
        else if (t.pageX < window.innerWidth * 0.5) this.onKey(true, { keyCode: 37 });
        else if (t.pageY > window.innerWidth * 0.5) this.onKey(true, { keyCode: 39 });
      };

      Controls.prototype.onTouchEnd = function(e) {
        this.states = { 'left': false, 'right': false, 'forward': false, 'backward': false };
        e.preventDefault();
        e.stopPropagation();
      };

      Controls.prototype.onKey = function(val, e) {
        var state = this.codes[e.keyCode];
        if (typeof state === 'undefined') return;
        this.states[state] = val;
        e.preventDefault && e.preventDefault();
        e.stopPropagation && e.stopPropagation();
      };

      function Bitmap(src, width, height) {
        this.image = new Image();
        this.image.src = src;
        this.width = width;
        this.height = height;
      }
      
      function Player(x, y, direction) {
        this.x = x;
        this.y = y;
        this.direction = direction;
        this.weapon = new Bitmap('assets/knife_hand.png', 319, 320);
        this.paces = 0;
      }

      Player.prototype.rotate = function(angle) {
        this.direction = (this.direction + angle + CIRCLE) % (CIRCLE);
      };

      Player.prototype.walk = function(distance, map) {
        var dx = Math.cos(this.direction) * distance;
        var dy = Math.sin(this.direction) * distance;
        if (map.get(this.x + dx, this.y) <= 0) this.x += dx;
        if (map.get(this.x, this.y + dy) <= 0) this.y += dy;
        this.paces += distance;
      };

      Player.prototype.update = function(controls, map, seconds) {
        if (controls.left) this.rotate(-Math.PI * seconds);
        if (controls.right) this.rotate(Math.PI * seconds);
        if (controls.forward) this.walk(3 * seconds, map);
        if (controls.backward) this.walk(-3 * seconds, map);
      };

      function Map(size) {
        this.size = size;
        this.wallGrid = new Uint8Array(size * size);
        this.skybox = new Bitmap('assets/deathvalley_panorama.jpg', 2000, 750);
        this.wallTexture = new Bitmap('assets/wall_texture.jpg', 1024, 1024);
        this.light = 0;
      }

      Map.prototype.get = function(x, y) {
        x = Math.floor(x);
        y = Math.floor(y);
        if (x < 0 || x > this.size - 1 || y < 0 || y > this.size - 1) return -1;
        return this.wallGrid[y * this.size + x];
      };

      Map.prototype.randomize = function() {
        for (var i = 0; i < this.size * this.size; i++) {
          this.wallGrid[i] = Math.random() < 0.3 ? 1 : 0;
        }
      };

      Map.prototype.cast = function(point, angle, range) {
        var self = this;
        var sin = Math.sin(angle);
        var cos = Math.cos(angle);
        var noWall = { length2: Infinity };

        return ray({ x: point.x, y: point.y, height: 0, distance: 0 });

        function ray(origin) {
          var stepX = step(sin, cos, origin.x, origin.y);
          var stepY = step(cos, sin, origin.y, origin.x, true);
          var nextStep = stepX.length2 < stepY.length2
            ? inspect(stepX, 1, 0, origin.distance, stepX.y)
            : inspect(stepY, 0, 1, origin.distance, stepY.x);

          if (nextStep.distance > range) return [origin];
          return [origin].concat(ray(nextStep));
        }

        function step(rise, run, x, y, inverted) {
          if (run === 0) return noWall;
          var dx = run > 0 ? Math.floor(x + 1) - x : Math.ceil(x - 1) - x;
          var dy = dx * (rise / run);
          return {
            x: inverted ? y + dy : x + dx,
            y: inverted ? x + dx : y + dy,
            length2: dx * dx + dy * dy
          };
        }

        function inspect(step, shiftX, shiftY, distance, offset) {
          var dx = cos < 0 ? shiftX : 0;
          var dy = sin < 0 ? shiftY : 0;
          step.height = self.get(step.x - dx, step.y - dy);
          step.distance = distance + Math.sqrt(step.length2);
          if (shiftX) step.shading = cos < 0 ? 2 : 0;
          else step.shading = sin < 0 ? 2 : 1;
          step.offset = offset - Math.floor(offset);
          return step;
        }
      };

      Map.prototype.update = function(seconds) {
        if (this.light > 0) this.light = Math.max(this.light - 10 * seconds, 0);
        else if (Math.random() * 5 < seconds) this.light = 2;
      };

      function Camera(canvas, resolution, focalLength) {
        this.ctx = canvas.getContext('2d');
        this.width = canvas.width = window.innerWidth * 0.5;
        this.height = canvas.height = window.innerHeight * 0.5;
        this.resolution = resolution;
        this.spacing = this.width / resolution;
        this.focalLength = focalLength || 0.8;
        this.range = MOBILE ? 8 : 14;
        this.lightRange = 5;
        this.scale = (this.width + this.height) / 1200;
      }

      Camera.prototype.render = function(player, map) {
        this.drawSky(player.direction, map.skybox, map.light);
        this.drawColumns(player, map);
        this.drawWeapon(player.weapon, player.paces);
      };

      Camera.prototype.drawSky = function(direction, sky, ambient) {
        var width = sky.width * (this.height / sky.height) * 2;
        var left = (direction / CIRCLE) * -width;

        this.ctx.save();
        this.ctx.drawImage(sky.image, left, 0, width, this.height);
        if (left < width - this.width) {
          this.ctx.drawImage(sky.image, left + width, 0, width, this.height);
        }
        if (ambient > 0) {
          this.ctx.fillStyle = '#ffffff';
          this.ctx.globalAlpha = ambient * 0.1;
          this.ctx.fillRect(0, this.height * 0.5, this.width, this.height * 0.5);
        }
        this.ctx.restore();
      };

      Camera.prototype.drawColumns = function(player, map) {
        this.ctx.save();
        for (var column = 0; column < this.resolution; column++) {
          var x = column / this.resolution - 0.5;
          var angle = Math.atan2(x, this.focalLength);
          var ray = map.cast(player, player.direction + angle, this.range);
          this.drawColumn(column, ray, angle, map);
        }
        this.ctx.restore();
      };

      Camera.prototype.drawWeapon = function(weapon, paces) {
        var bobX = Math.cos(paces * 2) * this.scale * 6;
        var bobY = Math.sin(paces * 4) * this.scale * 6;
        var left = this.width * 0.66 + bobX;
        var top = this.height * 0.6 + bobY;
        this.ctx.drawImage(weapon.image, left, top, weapon.width * this.scale, weapon.height * this.scale);
      };

      Camera.prototype.drawColumn = function(column, ray, angle, map) {
        var ctx = this.ctx;
        var texture = map.wallTexture;
        var left = Math.floor(column * this.spacing);
        var width = Math.ceil(this.spacing);
        var hit = -1;

        while (++hit < ray.length && ray[hit].height <= 0);

        for (var s = ray.length - 1; s >= 0; s--) {
          var step = ray[s];
          var rainDrops = Math.pow(Math.random(), 3) * s;
          var rain = (rainDrops > 0) && this.project(0.1, angle, step.distance);

          if (s === hit) {
            var textureX = Math.floor(texture.width * step.offset);
            var wall = this.project(step.height, angle, step.distance);

            ctx.globalAlpha = 1;
            ctx.drawImage(texture.image, textureX, 0, 1, texture.height, left, wall.top, width, wall.height);
            
            ctx.fillStyle = '#000000';
            ctx.globalAlpha = Math.max((step.distance + step.shading) / this.lightRange - map.light, 0);
            ctx.fillRect(left, wall.top, width, wall.height);
          }
          
          ctx.fillStyle = '#ffffff';
          ctx.globalAlpha = 0.15;
          while (--rainDrops > 0) ctx.fillRect(left, Math.random() * rain.top, 1, rain.height);
        }
      };

      Camera.prototype.project = function(height, angle, distance) {
        var z = distance * Math.cos(angle);
        var wallHeight = this.height * height / z;
        var bottom = this.height / 2 * (1 + 1 / z);
        return {
          top: bottom - wallHeight,
          height: wallHeight
        }; 
      };

      function GameLoop() {
        this.frame = this.frame.bind(this);
        this.lastTime = 0;
        this.callback = function() {};
      }

      GameLoop.prototype.start = function(callback) {
        this.callback = callback;
        requestAnimationFrame(this.frame);
      };

      GameLoop.prototype.frame = function(time) {
        var seconds = (time - this.lastTime) / 1000;
        this.lastTime = time;
        if (seconds < 0.2) this.callback(seconds);
        requestAnimationFrame(this.frame);
      };

      var display = document.getElementById('display');
      var player = new Player(15.3, -1.2, Math.PI * 0.3);
      var map = new Map(32);
      var controls = new Controls();
      var camera = new Camera(display, MOBILE ? 160 : 320, 0.8);
      var loop = new GameLoop();

      map.randomize();
      
      loop.start(function frame(seconds) {
        map.update(seconds);
        player.update(controls.states, map, seconds);
        camera.render(player, map);
      });

    </script>

function createSegment(index) {
    return {
        index: index,
        p1: { world: { x: 0, y: 0, z: index * SEGMENT_LENGTH }, camera: {}, screen: {} },
        p2: { world: { x: 0, y: 0, z: (index + 1) * SEGMENT_LENGTH }, camera: {}, screen: {} },
        curve: 0,
        sprites: [],
        color: Math.floor(index / RUMBLE_LENGTH) % 2
    };
}

function addSegment(curve, y) {
    const index = segments.length;
    const segment = createSegment(index);
    segment.curve = curve;
    segment.p1.world.y = segments.length > 0 ? segments[segments.length - 1].p2.world.y : 0;
    segment.p2.world.y = y;
    
    if (Math.random() < 0.08 && index > 20) {
        segment.sprites.push({ type: 'tree', offset: -2.5 - Math.random() * 6 });
    }
    if (Math.random() < 0.03 && index > 20) {
        segment.sprites.push({ type: 'rock', offset: 2.5 + Math.random() * 6 });
    }
    if (Math.random() < 0.015 && index > 50) {
        segment.sprites.push({ type: 'car', offset: (Math.random() * 1.6) - 0.8 });
    }
    
    segments.push(segment);
}

function easeInOut(a, b, t) {
    return a + (b - a) * ((-Math.cos(t * Math.PI) / 2) + 0.5);
}

function addRoad(enter, hold, leave, curve, height) {
    const startY = segments.length > 0 ? segments[segments.length - 1].p2.world.y : 0;
    const endY = startY + height * SEGMENT_LENGTH;
    const total = enter + hold + leave;
    
    for (let i = 0; i < enter; i++) {
        const t = i / enter;
        const c = easeInOut(0, curve, t);
        const y = easeInOut(startY, endY, i / total);
        addSegment(c, y);
    }
    for (let i = 0; i < hold; i++) {
        const t = (enter + i) / total;
        const y = easeInOut(startY, endY, t);
        addSegment(curve, y);
    }
    for (let i = 0; i < leave; i++) {
        const t = i / leave;
        const c = easeInOut(curve, 0, t);
        const y = easeInOut(startY, endY, (enter + hold + i) / total);
        addSegment(c, y);
    }
}

        // ОЧЕНЬ плавный левый поворот
        () => addRoad(40, 60, 40, -0.4 - Math.random() * 0.3, 0),
        // ОЧЕНЬ плавный правый поворот
        () => addRoad(40, 60, 40, 0.4 + Math.random() * 0.3, 0),
        // Небольшая горка (без поворота)
        () => addRoad(40, 50, 40, 0, 10 + Math.random() * 10),
        // Небольшой спуск
        () => addRoad(40, 50, 40, 0, -10 - Math.random() * 10),
        // ОЧЕНЬ плавный S-поворот (редко)

            addRoad(30, 40, 30, 0.5, 0);

        // Длинная прямая с небольшим подъемом
        () => addRoad(50, 100, 50, 0, 5),

    const count = 2 + Math.floor(Math.random() * 2);

        patterns[Math.floor(Math.random() * patterns.length)]();
    }
}

function drawSprite(ctx, type, x, y, scale) {
    const w = 300 * scale * ROAD_WIDTH * 0.002;
    const h = 300 * scale * ROAD_WIDTH * 0.002;

        ctx.moveTo(destX + w/2, destY);
        ctx.lineTo(destX + w, destY + h*0.5);

}

function project(p, cameraX, cameraY, cameraZ, width, height) {
    p.camera.x = p.world.x - cameraX;
    p.camera.y = p.world.y - cameraY;
    p.camera.z = p.world.z - cameraZ;
    p.screen.scale = CAMERA_DEPTH / p.camera.z;
    p.screen.x = Math.round(width/2 + p.screen.scale * p.camera.x * width/2);
    p.screen.y = Math.round(height/2 - p.screen.scale * p.camera.y * height/2);
    p.screen.w = Math.round(p.screen.scale * ROAD_WIDTH * width/2);
}

function renderPolygon(x1, y1, x2, y2, x3, y3, x4, y4, color) {
    ctx.fillStyle = color;
    ctx.beginPath();
    ctx.moveTo(x1, y1);
    ctx.lineTo(x2, y2);
    ctx.lineTo(x3, y3);
    ctx.lineTo(x4, y4);
    ctx.closePath();
    ctx.fill();
}

function render() {
    ctx.clearRect(0, 0, width, height);
    
    const baseIndex = Math.floor(playerZ / SEGMENT_LENGTH);
    const baseSegment = segments[baseIndex % segments.length];
    const basePercent = (playerZ % SEGMENT_LENGTH) / SEGMENT_LENGTH;
    const playerY = baseSegment.p1.world.y + (baseSegment.p2.world.y - baseSegment.p1.world.y) * basePercent;
    
    let dx = -baseSegment.curve * basePercent;
    let x = 0;
    let maxY = height;
    
    renderBackground(baseSegment.curve);
    
    for (let n = 0; n < DRAW_DISTANCE; n++) {
        const segmentIndex = (baseIndex + n) % segments.length;
        const segment = segments[segmentIndex];
        
        const cameraX = playerX * ROAD_WIDTH;
        const cameraY = CAMERA_HEIGHT + playerY;
        const cameraZ = playerZ;
        
        project(segment.p1, cameraX, cameraY, cameraZ, width, height);
        project(segment.p2, cameraX, cameraY, cameraZ, width, height);
        
        x += dx;
        dx += segment.curve;
        
        segment.p1.screen.x += x;
        segment.p2.screen.x += x;
        
        if (segment.p1.camera.z <= CAMERA_DEPTH || 
            segment.p2.screen.y >= maxY || 
            segment.p2.screen.y >= segment.p1.screen.y) {
            continue;
        }
        
        const colorIdx = segment.color;
        const colors = {
            grass: colorIdx ? '#2E7D32' : '#1B5E20',
            rumble: colorIdx ? '#FFF' : '#C62828',
            road: colorIdx ? '#757575' : '#616161',
            lane: colorIdx ? '#FFF' : null
        };
        
        const r1 = segment.p1.screen.w / 32;
        const r2 = segment.p2.screen.w / 32;
        
        ctx.fillStyle = colors.grass;
        ctx.fillRect(0, segment.p2.screen.y, width, segment.p1.screen.y - segment.p2.screen.y);
        
        renderPolygon(
            segment.p1.screen.x - segment.p1.screen.w - r1, segment.p1.screen.y,
            segment.p1.screen.x - segment.p1.screen.w, segment.p1.screen.y,
            segment.p2.screen.x - segment.p2.screen.w, segment.p2.screen.y,
            segment.p2.screen.x - segment.p2.screen.w - r2, segment.p2.screen.y,
            colors.rumble
        );
        
        renderPolygon(
            segment.p1.screen.x + segment.p1.screen.w + r1, segment.p1.screen.y,
            segment.p1.screen.x + segment.p1.screen.w, segment.p1.screen.y,
            segment.p2.screen.x + segment.p2.screen.w, segment.p2.screen.y,
            segment.p2.screen.x + segment.p2.screen.w + r2, segment.p2.screen.y,
            colors.rumble
        );
        
        renderPolygon(
            segment.p1.screen.x - segment.p1.screen.w, segment.p1.screen.y,
            segment.p1.screen.x + segment.p1.screen.w, segment.p1.screen.y,
            segment.p2.screen.x + segment.p2.screen.w, segment.p2.screen.y,
            segment.p2.screen.x - segment.p2.screen.w, segment.p2.screen.y,
            colors.road
        );
        
        if (colors.lane) {
            const laneW1 = segment.p1.screen.w * 2 / LANES;
            const laneW2 = segment.p2.screen.w * 2 / LANES;
            let laneX1 = segment.p1.screen.x - segment.p1.screen.w + laneW1;
            let laneX2 = segment.p2.screen.x - segment.p2.screen.w + laneW2;
            
            for (let lane = 1; lane < LANES; lane++) {
                renderPolygon(
                    laneX1 - segment.p1.screen.w/30, segment.p1.screen.y,
                    laneX1 + segment.p1.screen.w/30, segment.p1.screen.y,
                    laneX2 + segment.p2.screen.w/30, segment.p2.screen.y,
                    laneX2 - segment.p2.screen.w/30, segment.p2.screen.y,
                    colors.lane
                );
                laneX1 += laneW1;
                laneX2 += laneW2;
            }
        }
        
        maxY = segment.p1.screen.y;
    }
    
    for (let n = DRAW_DISTANCE - 1; n > 0; n--) {
        const segmentIndex = (baseIndex + n) % segments.length;
        const segment = segments[segmentIndex];
        
        for (const sprite of segment.sprites) {
            const spriteScale = segment.p1.screen.scale;
            const spriteX = segment.p1.screen.x + spriteScale * sprite.offset * ROAD_WIDTH * width / 2;
            const spriteY = segment.p1.screen.y;
            
            if (spriteX > -300 && spriteX < width + 300) {
                drawSprite(ctx, sprite.type, spriteX, spriteY, spriteScale);
            }
        }
    }
    
    drawPlayer(ctx, width, height, keyLeft ? -1 : (keyRight ? 1 : 0));
}

function update(dt) {
    const speedPercent = speed / MAX_SPEED;
    const dx = dt * 2 * speedPercent;
    
    playerZ += speed * dt;
    totalDist += speed * dt;
    
    const segmentsAhead = segments.length - Math.floor(playerZ / SEGMENT_LENGTH);
    if (segmentsAhead < DRAW_DISTANCE + 200) {
        generateRoad();
    }
    
    if (keyLeft) playerX -= dx;
    if (keyRight) playerX += dx;
    
    if (playerX < -2.5) playerX = -2.5;
    if (playerX > 2.5) playerX = 2.5;
    
    if (speed < MAX_SPEED) speed += ACCEL * dt;
    
    if ((playerX < -1 || playerX > 1) && speed > MAX_SPEED / 4) {
        speed -= ACCEL * dt * 1.5;
    }
    
    document.getElementById('speedVal').textContent = Math.floor(speed / 100);
    document.getElementById('distVal').textContent = Math.floor(totalDist / 10000);
}

function readCsvRows(): array {
    $rows = [];
    if (!file_exists(CSV_FILE)) return $rows;
    $fh = fopen(CSV_FILE, 'r');
    fgetcsv($fh); // заголовок
    while (($row = fgetcsv($fh)) !== false) {
        if (count($row) < 5) continue;
        $rows[] = [
            'name'        => $row[0],
            'uploaded_at' => $row[1],
            'type'        => $row[2],
            'size'        => (int)$row[3],
            'downloads'   => (int)$row[4],
        ];
    }
    fclose($fh);
    return $rows;
}

function writeCsvRows(array $rows): void {
    $fh = fopen(CSV_FILE, 'w');
    flock($fh, LOCK_EX);
    fputcsv($fh, ['name', 'uploaded_at', 'type', 'size', 'downloads']);
    foreach ($rows as $r) {
        fputcsv($fh, [$r['name'], $r['uploaded_at'], $r['type'], $r['size'], $r['downloads']]);
    }
    flock($fh, LOCK_UN);
    fclose($fh);
}

function appendCsvRow(array $r): void {
    $fh = fopen(CSV_FILE, 'a');
    flock($fh, LOCK_EX);
    fputcsv($fh, [$r['name'], $r['uploaded_at'], $r['type'], $r['size'], $r['downloads']]);
    flock($fh, LOCK_UN);
    fclose($fh);
}

function humanSize(int $bytes): string {
    $units = ['B', 'KB', 'MB', 'GB', 'TB'];
    $i = 0;
    while ($bytes >= 1024 && $i < count($units) - 1) {
        $bytes /= 1024; $i++;
    }
    return round($bytes, 2) . ' ' . $units[$i];
}

<?php
require __DIR__ . '/config.php';
header('Content-Type: application/json');

try {
    if ($_SERVER['REQUEST_METHOD'] !== 'POST') {
        throw new Exception('Метод не поддерживается');
    }

    $fileId   = $_POST['fileId']   ?? '';
    $fileName = $_POST['fileName'] ?? '';
    //$chunkIdx = (int)($_POST['chunk'] ?? 0);
	$chunkIdx = (int)($_POST['chunkIndex'] ?? 0);
    $total    = (int)($_POST['totalChunks'] ?? 1);

    if (!preg_match('/^[a-f0-9]{16,64}$/', $fileId)) {
        throw new Exception('Неверный идентификатор');
    }
    if ($fileName === '') {
        throw new Exception('Имя файла не указано');
    }

    $safeName = basename($fileName);

    // Копим чанки во временной папке
    $chunkDir = CHUNK_DIR . $fileId . '/';
    if (!is_dir($chunkDir)) mkdir($chunkDir, 0775, true);

    if (empty($_FILES['chunk']) || $_FILES['chunk']['error'] !== UPLOAD_ERR_OK) {
        throw new Exception('Ошибка получения чанка');
    }

    $partPath = $chunkDir . 'part_' . $chunkIdx;
    if (!move_uploaded_file($_FILES['chunk']['tmp_name'], $partPath)) {
        throw new Exception('Не удалось сохранить чанк');
    }

    // Проверяем, все ли чанки на месте
    $received = count(glob($chunkDir . 'part_*'));
    if ($received < $total) {
        echo json_encode([
            'ok'      => true,
            'done'    => false,
            'received'=> $received,
            'total'   => $total,
        ]);
        exit;
    }

    // Склейка в один файл
    $finalName = $safeName;
    $target = UPLOAD_DIR . $finalName;
    // Уникальное имя, если файл уже есть
    if (file_exists($target)) {
        $info = pathinfo($safeName);
        $finalName = $info['filename'] . '_' . time() . '.' . ($info['extension'] ?? '');
        $target = UPLOAD_DIR . $finalName;
    }

    $out = fopen($target, 'wb');
    for ($i = 0; $i < $total; $i++) {
        $p = $chunkDir . 'part_' . $i;
        if (!file_exists($p)) { fclose($out); throw new Exception('Пропал чанк ' . $i); }
        $in = fopen($p, 'rb');
        stream_copy_to_stream($in, $out);
        fclose($in);
        unlink($p);
    }
    fclose($out);
    @rmdir($chunkDir);

    $size = filesize($target);
    $ext  = strtolower(pathinfo($finalName, PATHINFO_EXTENSION));

    appendCsvRow([
        'name'        => $finalName,
        'uploaded_at' => date('Y-m-d H:i:s'),
        'type'        => $ext ?: 'unknown',
        'size'        => $size,
        'downloads'   => 0,
    ]);

    echo json_encode([
        'ok'   => true,
        'done' => true,
        'name' => $finalName,
        'size' => humanSize($size),
    ]);

} catch (Throwable $e) {
    http_response_code(400);
    echo json_encode(['ok' => false, 'error' => $e->getMessage()]);
}

function readScores($file) {
    $scores = [];
    if (file_exists($file) && ($handle = fopen($file, "r")) !== FALSE) {
        while (($data = fgetcsv($handle, 1000, ";")) !== FALSE) {
            if (count($data) >= 2) {
                $scores[trim($data[0])] = (int)trim($data[1]);
            }
        }
        fclose($handle);

}

===== localhost/video.play/php/scan.php =====
<?php
define('UPLOAD_DIR', __DIR__ . '/../upload/');
define('DB_FILE',    __DIR__ . '/../res/db.csv');

function ensureDirs() {
    if (!is_dir(UPLOAD_DIR)) mkdir(UPLOAD_DIR, 0777, true);
    if (!is_dir(dirname(DB_FILE))) mkdir(dirname(DB_FILE), 0777, true);
}

function loadDb() {
    ensureDirs();
    if (!file_exists(DB_FILE)) return [];
    $fp = fopen(DB_FILE, 'r');
    if (!$fp) return [];
    $header = fgetcsv($fp);
    $rows = [];
    while (($r = fgetcsv($fp)) !== false) {
        if (count($r) < 10) continue;
        $rows[] = [
            'id'       => $r[0],
            'name'     => $r[1],
            'path'     => $r[2],
            'filename' => $r[3],
            'size'     => (int)$r[4],
            'date'     => $r[5],
            'added'    => (int)$r[6],
            'views'    => (int)$r[7],
            'likes'    => (int)$r[8],
            'dislikes' => (int)$r[9],
        ];
    }
    fclose($fp);
    return $rows;
}

function saveDb(array $rows) {
    ensureDirs();
    usort($rows, fn($a,$b) => $b['added'] <=> $a['added']);
    $fp = fopen(DB_FILE, 'w');
    fputcsv($fp, ['id','name','path','filename','size','date','added','views','likes','dislikes']);
    foreach ($rows as $r) {
        fputcsv($fp, [
            $r['id'], $r['name'], $r['path'], $r['filename'],
            $r['size'], $r['date'], $r['added'],
            $r['views'], $r['likes'], $r['dislikes']
        ]);
    }
    fclose($fp);
}

function syncDatabase() {
    ensureDirs();
    $db = loadDb();
    $byFile = [];
    foreach ($db as $r) $byFile[$r['filename']] = $r;

    $real = [];
    // Поддерживаемые расширения
    $allowedExts = ['mp4', 'mkv', 'avi', 'webm', 'mov', 'flv', 'wmv', 'm4v', '3gp', 'ogv', 'mpg', 'mpeg', 'ts'];
    $files = [];
    foreach ($allowedExts as $ext) {
        $files = array_merge($files, glob(UPLOAD_DIR . '*.' . $ext));
        $files = array_merge($files, glob(UPLOAD_DIR . '*.' . strtoupper($ext)));
    }
    
    foreach ($files as $full) {
        $fn = basename($full);
        $real[$fn] = true;
        $relPath = 'upload/' . $fn;

        if (isset($byFile[$fn])) {
            $byFile[$fn]['size'] = filesize($full);
            $byFile[$fn]['date'] = date('Y-m-d H:i', filemtime($full));
            $byFile[$fn]['path'] = $relPath;
        } else {
            $db[] = [
                'id'       => bin2hex(random_bytes(6)),
                'name'     => pathinfo($fn, PATHINFO_FILENAME),
                'path'     => $relPath,
                'filename' => $fn,
                'size'     => filesize($full),
                'date'     => date('Y-m-d H:i', filemtime($full)),
                'added'    => time(),
                'views'    => 0,
                'likes'    => 0,
                'dislikes' => 0,
            ];
        }
    }

    $db = array_values(array_filter($db, fn($r) => isset($real[$r['filename']])));
    saveDb($db);
}

===== localhost/video.play/php/upload.php =====
<?php
require __DIR__ . '/scan.php';

header('Content-Type: application/json; charset=utf-8');

function fail($msg, $code = 400) {
    http_response_code($code);
    echo json_encode(['ok' => false, 'error' => $msg], JSON_UNESCAPED_UNICODE);
    exit;
}

if ($_SERVER['REQUEST_METHOD'] !== 'POST') {
    fail('method not allowed', 405);
}

$action = $_POST['action'] ?? '';

$tempDir = UPLOAD_DIR . 'chunks/';
if (!is_dir($tempDir)) {
    mkdir($tempDir, 0777, true);
}

if ($action === 'chunk') {
    if (!isset($_FILES['chunk']) || !is_array($_FILES['chunk'])) {
        fail('кусок не получен');
    }
    $f = $_FILES['chunk'];
    if ($f['error'] !== UPLOAD_ERR_OK) {
        fail('ошибка куска: код ' . $f['error']);
    }
    $fileId = preg_replace('/[^a-zA-Z0-9_-]/', '', $_POST['fileId'] ?? '');
    if (!$fileId) fail('не передан fileId');
    $chunkIndex = (int)($_POST['chunkIndex'] ?? 0);
    $chunkPath = $tempDir . $fileId . '_' . $chunkIndex . '.part';
    if (!move_uploaded_file($f['tmp_name'], $chunkPath)) {
        fail('не удалось сохранить кусок', 500);
    }
    echo json_encode(['ok' => true]);
    exit;
}

if ($action === 'finalize') {
    $fileId = preg_replace('/[^a-zA-Z0-9_-]/', '', $_POST['fileId'] ?? '');
    $fileName = $_POST['fileName'] ?? 'video.mp4';
    $totalChunks = (int)($_POST['totalChunks'] ?? 0);

    if (!$fileId || !$totalChunks) fail('неверные параметры финализации');

    $ext = strtolower(pathinfo($fileName, PATHINFO_EXTENSION));
    $allowed = ['mp4', 'mkv', 'avi', 'webm', 'mov', 'flv', 'wmv', 'm4v', '3gp', 'ogv', 'mpg', 'mpeg', 'ts'];
    if (!in_array($ext, $allowed)) fail('Недопустимый формат видео');
    
    $base = pathinfo($fileName, PATHINFO_FILENAME);
    $base = preg_replace('/[^\p{L}\p{N}_\- ]+/u', '_', $base);
    $base = trim($base, ' _-');
    if ($base === '') $base = 'video';

    // Временный файл (оригинал)
    $tempTarget = $tempDir . $fileId . '_merged.' . $ext;

    $out = fopen($tempTarget, 'wb');
    if (!$out) fail('не удалось создать временный файл', 500);

    for ($i = 0; $i < $totalChunks; $i++) {
        $chunkPath = $tempDir . $fileId . '_' . $i . '.part';
        if (!file_exists($chunkPath)) {
            fclose($out);
            @unlink($tempTarget);
            fail("пропал кусок #$i");
        }
        $in = fopen($chunkPath, 'rb');
        while (!feof($in)) {
            fwrite($out, fread($in, 1024 * 1024));
        }
        fclose($in);
    }
    fclose($out);

    // Удаляем куски
    for ($i = 0; $i < $totalChunks; $i++) {
        @unlink($tempDir . $fileId . '_' . $i . '.part');
    }

    // --- КОНВЕРТАЦИЯ ---
    // Финальный файл всегда будет .mp4 для совместимости
    $target = UPLOAD_DIR . $base . '.mp4';
    $i = 1;
    while (file_exists($target)) {
        $target = UPLOAD_DIR . $base . '_' . $i . '.mp4';
        $i++;
    }

    // Ищем FFmpeg
    $ffmpeg = null;
    $possiblePaths = [
        'C:\OSPanel\modules\ffmpeg\ffmpeg.exe',
        'C:\OSPanel\modules\ffmpeg\bin\ffmpeg.exe',
        'ffmpeg'
    ];
    
    foreach ($possiblePaths as $p) {
        if (strpos($p, ':') !== false) {
            if (file_exists($p)) { $ffmpeg = $p; break; }
        } else {
            // Если это просто "ffmpeg", проверяем через exec
            exec("where $p", $o, $r);
            if ($r === 0) { $ffmpeg = $p; break; }
        }
    }

    if (!$ffmpeg) {
        @file_put_contents(__DIR__ . '/../res/ffmpeg_error.log', "FFmpeg не найден!\n", FILE_APPEND);
        // Если ffmpeg нет, просто сохраняем как есть
        $target = UPLOAD_DIR . $base . '.' . $ext;
        @rename($tempTarget, $target);
    } else {
        // Если это mp4, но нужно перекодировать звук (на всякий случай)
        // Но обычно mp4 уже норм.
        // Для mkv/avi/flv/wmv и т.д. — перекодируем ВСЕГДА.
        
        if ($ext !== 'mp4') {
            // Команда: видео H264, звук AAC (стерео, 128кб/с)
            // -movflags +faststart ускоряет старт
            $cmd = escapeshellarg($ffmpeg) . " -y -i " . escapeshellarg($tempTarget) . " -c:v libx264 -preset fast -crf 23 -c:a aac -b:a 128k -ac 2 -movflags +faststart " . escapeshellarg($target) . " 2>&1";
            
            exec($cmd, $output, $returnCode);

            if ($returnCode !== 0 || !file_exists($target) || filesize($target) < 1000) {
                // Ошибка конвертации — сохраняем оригинал
                @file_put_contents(__DIR__ . '/../res/ffmpeg_error.log', date('Y-m-d H:i:s') . " Ошибка: $cmd\n" . implode("\n", $output) . "\n\n", FILE_APPEND);
                $target = UPLOAD_DIR . $base . '.' . $ext;
                @rename($tempTarget, $target);
            } else {
                // Успех — оригинал удаляем
                @unlink($tempTarget);
            }
        } else {
            // Если это mp4, просто переносим
            @rename($tempTarget, $target);
        }
    }

    syncDatabase();

    $db = loadDb();
    $newId = null;
    $fn = basename($target);
    foreach ($db as $r) {
        if ($r['filename'] === $fn) { $newId = $r['id']; break; }
    }

    echo json_encode(['ok' => true, 'id' => $newId, 'file' => $fn], JSON_UNESCAPED_UNICODE);
    exit;
}
